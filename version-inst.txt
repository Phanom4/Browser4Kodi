# Qellen
wget -O Browser4Kodi-addon.zip \
  https://github.com/Phanom4/Browser4Kodi/releases/download/v0.0.2-test/Browser4Kodi-addon.zip
unzip  Browser4Kodi-addon.zip
cd Browser4Kodi/k21_3_c150 
rm -rf cef chromium build depot_tools 
git clone --branch 7871 --single-branch https://github.com/chromiumembedded/cef.git cef
wget https://raw.githubusercontent.com/chromiumembedded/cef/master/tools/automate/automate-git.py
sed -i "s/sync_args = '--nohooks '/sync_args = '--nohooks --jobs=1 '/" automate-git.py
python3 automate-git.py --download-dir="$(pwd)" --branch=7871 --no-build --no-distrib --no-chromium-history --x64-build

GN_DEFINES="is_official_build=true use_sysroot=true symbol_level=1 is_cfi=false use_thin_lto=false chrome_pgo_phase=0" \
python3 automate-git.py \
  --download-dir="$(pwd)" \
  --branch=7871 \
  --no-update \
  --force-build \
  --no-debug-build \
  --build-target=cefsimple \
  --x64-build \
  --no-chromium-history
