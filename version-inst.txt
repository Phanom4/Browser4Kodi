# ======================
# Bitte Beachten
# Download der Quellen: über 30 Minuten bei 100 mbits Telekom
# Kompilierung von CEF: über 40 Minuten bei 32 kerne 64 threads 64 gb Ram
# Kompilierung von Browser4Kodi: X Minuten bei 32 kerne 64 threads 64 gb Ram
# Betriebssystem Gentoo

# Qellen
mkdir Browser4Kodi
cd Browser4Kodi 
git clone https://github.com/Phanom4/Browser4Kodi.git .
rm -rf cef chromium build
git clone --branch 7871 --single-branch https://github.com/chromiumembedded/cef.git cef
wget https://raw.githubusercontent.com/chromiumembedded/cef/master/tools/automate/automate-git.py
sed -i "s/sync_args = '--nohooks '/sync_args = '--nohooks --jobs=1 '/" automate-git.py
python3 automate-git.py --download-dir="$(pwd)" --branch=7871 --no-build --no-distrib --no-chromium-history --x64-build

GN_DEFINES="is_official_build=true use_sysroot=true symbol_level=1 is_cfi=false use_thin_lto=false chrome_pgo_phase=0" \
python3 automate-git.py \
  --download-dir="$(pwd)" \
  --branch=7871 \
  --no-update \
  --force-build \
  --no-debug-build \
  --build-target=cefsimple \
  --x64-build \
  --no-chromium-history

#Browser4Kodi
cmake -S . -B build && cmake --build build -j 64

#install
pyton browser_k21_3_c150_install_0_1.py -1
